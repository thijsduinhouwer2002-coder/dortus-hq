===== itch / Controles (naam, versie, build, akkoord Daan) (laatste 58 van 58 regels) =====
##[group]Run S="$GITHUB_STEP_SUMMARY"; fout=0
S="$GITHUB_STEP_SUMMARY"; fout=0
naam="${GITHUB_REF_NAME#uitgave/}"
if [ "$naam" != "$GITHUB_REF_NAME" ]; then
  if [[ "$naam" =~ ^hats-itch-v([0-9]+)\.([0-9]+)\.([0-9]+)(-proef[a-z0-9]*)?$ ]]; then
    versie="${BASH_REMATCH[1]}.${BASH_REMATCH[2]}.${BASH_REMATCH[3]}"
    [ -n "${BASH_REMATCH[4]}" ] && proef=true || proef=false
  else
    echo "::error::Geen geldige naam '$GITHUB_REF_NAME'. Vorm: uitgave/hats-itch-v1.2.3 of uitgave/hats-itch-v1.2.3-proef1"; exit 1
  fi
else
  versie=""; proef="$IN_PROEF"
fi
meld() { if [ "$proef" = "true" ]; then echo "::warning::$1"; echo "- waarschuwing: $1" >> "$S";
         else echo "::error::$1"; echo "- FOUT: $1" >> "$S"; fout=$((fout+1)); fi; }
echo "## Controles ($( [ "$proef" = "true" ] && echo proefdraai || echo ECHTE UPLOAD ), $GITHUB_REF_NAME)" >> "$S"

# a. versie
pv=$(grep -o 'application/product_version="[^"]*"' producten/steam-hol/godot/export_presets.cfg | cut -d'"' -f2 | sort -u | head -1 | cut -d. -f1-3)
[ -n "$versie" ] || versie="$pv"
[ "$versie" = "$pv" ] || meld "taknaam zegt $versie, export_presets.cfg zegt $pv"
echo "- versie: $versie" >> "$S"

# b. geslaagde exe-run op main met dezelfde spelcode
boom=$(git rev-parse "$GITHUB_SHA:producten/steam-hol")
curl -fsS -H "[weggehaald: Authorization-kop]
  "https://api.github.com/repos/$GITHUB_REPOSITORY/actions/workflows/steam-hol-exe.yml/runs?branch=main&status=success&per_page=30" \
  -o runs.json || echo '{"workflow_runs":[]}' > runs.json
run=""
for r in $(jq -r '.workflow_runs[] | "\(.head_sha),\(.id)"' runs.json); do
  sha=${r%,*}; id=${r#*,}
  git cat-file -e "$sha^{commit}" 2>/dev/null || continue
  [ "$(git rev-parse "$sha:producten/steam-hol" 2>/dev/null)" = "$boom" ] || continue
  n=$(curl -fsS -H "[weggehaald: Authorization-kop]
      | jq '[.artifacts[] | select(.name=="speeltest-hats-of-the-fallen-windows" and .expired==false)] | length')
  if [ "${n:-0}" -gt 0 ]; then run="$id"; break; fi
done
if [ -n "$run" ]; then echo "- build: steam-hol-exe.yml run $run (zelfde spelcode, boom ${boom:0:7})" >> "$S"
else echo "::error::geen geslaagde steam-hol-exe.yml-run op main met deze spelcode en een levend speeltest-artifact (boom ${boom:0:7}). Start die eerst."; exit 1; fi

# c. akkoord Daan
rij=$(grep -E "^\| *$versie *\|" platformen/steam/uitgaven.md 2>/dev/null | head -1 || true)
keuring=$(echo "$rij" | awk -F'|' '{gsub(/[ `]/,"",$4); print $4}')
oordeel=$(echo "$rij" | awk -F'|' '{gsub(/ /,"",$3); print tolower($3)}')
if [ "$oordeel" = "ja" ] && [ -n "$keuring" ] && [ -f "$keuring" ]; then echo "- akkoord Daan: ja ($keuring)" >> "$S"
else meld "geen akkoord van Daan voor $versie in platformen/steam/uitgaven.md"; fi

echo "proef=$proef" >> "$GITHUB_OUTPUT"; echo "versie=$versie" >> "$GITHUB_OUTPUT"; echo "run=$run" >> "$GITHUB_OUTPUT"
echo "uitslag controles: $fout FOUT" | tee -a "$S"
[ "$fout" -eq 0 ] || exit 1
shell: /usr/bin/bash --noprofile --norc -e -o pipefail {0}
env:

  GH_TOKEN: ***
  IN_PROEF:
