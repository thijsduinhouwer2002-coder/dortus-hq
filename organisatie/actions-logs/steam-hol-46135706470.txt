$uit = (Resolve-Path "producten/steam-hol/e2e/screenshots").Path -replace '\\','/'
$inst = Join-Path $env:APPDATA "Dortus/Het Hol/instellingen.json"
function Start1($log, $args_) {
  Remove-Item $inst -ErrorAction SilentlyContinue
  $p = Start-Process -FilePath "godot/Godot_v4.3-stable_win64.exe" `
    -ArgumentList "--path producten/steam-hol/godot --rendering-driver opengl3_angle -- vers=1 duur=6 $args_ log=$log" -PassThru
  if (-not $p.WaitForExit(90000)) { Stop-Process -Id $p.Id -Force }
  (Get-Content $log -ErrorAction SilentlyContinue | Select-String "HOL eerste start|HOL start|SCRIPT ERROR") -join "`n"
}
$fout = 0
$a = Start1 "$uit/schaal-gewoon.log" ""
Write-Output "===== gewoon"; Write-Output $a
if ($a -notmatch "HOL eerste start .*dpi=(\d+) hoogte=(200|240)") { Write-Output "FOUT: geen eerste-startregel"; $fout++ }
$b = Start1 "$uit/schaal-150.log" "dpi=144"
Write-Output "===== dpi=144 (150%)"; Write-Output $b
if ($b -notmatch "HOL eerste start .*dpi=144 hoogte=240" -or $b -notmatch "HOL start hoogte=240 .*venster=\(\d+, 240\)") { Write-Output "FOUT: op 150% niet 240 px (#333)"; $fout++ }
$c = Start1 "$uit/schaal-gekozen.log" "dpi=144 hoogte=96"
Write-Output "===== dpi=144 hoogte=96"; Write-Output $c
if ($c -match "HOL eerste start" -or $c -notmatch "HOL start hoogte=96 ") { Write-Output "FOUT: gekozen hoogte niet gevolgd"; $fout++ }
if ("$a$b$c" -match "SCRIPT ERROR") { Write-Output "FOUT: SCRIPT ERROR bij een eerste start"; $fout++ }
Write-Output "schaal: $fout FOUT"
if ($fout -gt 0) { exit 1 }
