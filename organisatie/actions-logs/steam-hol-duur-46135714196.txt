##[group]Run $uit = "producten/steam-hol/e2e/screenshots/duur"
$uit = "producten/steam-hol/e2e/screenshots/duur"
New-Item -ItemType Directory -Force $uit | Out-Null
$vol = (Resolve-Path $uit).Path
Start-Process notepad.exe
Start-Sleep 3
$sec = [int]$env:MINUTEN * 60
$log = ($vol -replace '\\','/') + "/hol.log"
# Het spel stopt zelf na duur+30 s; wij meten tot duur.
$glog = Join-Path $env:APPDATA "Dortus/Het Hol/logs/godot.log"
if ($env:EXE -eq 'ja') {
  # de echte .exe: geen --path, het project zit in de .pck naast de exe
  $prog = (Resolve-Path "producten/steam-hol/bouw/volledig/HetHol.exe").Path
  $voor = ""
  Remove-Item $glog -ErrorAction SilentlyContinue
} else {
  $prog = "godot/Godot_v4.3-stable_win64.exe"
  $voor = "--path producten/steam-hol/godot "
}
Write-Output "SOORT: $(if ($env:EXE -eq 'ja') { 'echte .exe (release)' } else { 'editor-Godot (--path)' })"
$p = Start-Process -FilePath $prog `
  -ArgumentList "$($voor)--rendering-driver opengl3_angle -- hoogte=120 vers=1 duur=$($sec + 30) $env:EXTRA log=$log" `
  -RedirectStandardOutput "$vol/uit.txt" -RedirectStandardError "$vol/fout.txt" -PassThru
$null = $p.Handle
Start-Sleep 20
if ($p.HasExited) { Write-Output "PROCES GESTOPT binnen 20 s, code $($p.ExitCode)"; Get-Content "$vol/uit.txt", "$vol/fout.txt" -ErrorAction SilentlyContinue; exit 1 }
$metingen = @()
$p.Refresh(); $vorig = $p.TotalProcessorTime.TotalSeconds
$stappen = [int](($sec - 20) / 10)
$dood = $false
for ($i = 1; $i -le $stappen; $i++) {
  Start-Sleep 10
  if ($p.HasExited) { $dood = $true; Write-Output "PROCES GESTOPT na $(20 + $i * 10) s, code $($p.ExitCode)"; break }
  $p.Refresh(); $nu = $p.TotalProcessorTime.TotalSeconds
  $metingen += [math]::Round($nu - $vorig, 3); $vorig = $nu
  if ($i % 60 -eq 0) {
    & producten/steam-hol/e2e/scherm.ps1 "$vol/strook-$([int]($i / 6))min.png"
    Write-Output "na $([int]($i / 6)) min: geheugen $([int]($p.WorkingSet64 / 1MB)) MB, laatste 10 s $($metingen[-1]) CPU-s"
  }
}
$gesorteerd = $metingen | Sort-Object
$gem = if ($metingen.Count) { [math]::Round(($metingen | Measure-Object -Average).Average, 3) } else { -1 }
$med = if ($metingen.Count) { $gesorteerd[[int]($gesorteerd.Count / 2)] } else { -1 }
$samen = @(
  "metingen van 10 s: $($metingen.Count)",
  "CPU-s per 10 s: gemiddeld $gem, mediaan $med, laagste $($gesorteerd[0]), hoogste $($gesorteerd[-1])",
  "geheugen aan het eind: $(if (-not $p.HasExited) { [int]($p.WorkingSet64 / 1MB) } else { 'proces weg' }) MB",
  "proces leefde tot het eind: $(-not $dood)",
  "soort: $(if ($env:EXE -eq 'ja') { 'echte .exe (release)' } else { 'editor-Godot (--path)' })",
  "eerste koop mogelijk: $((Select-String -Path ($vol + '/hol.log') -Pattern 'HOL eerste koop mogelijk.*' -ErrorAction SilentlyContinue | Select-Object -First 1).Matches.Value)"
)
# Iris keuring 9 (16 nov): hoe vaak dezelfde kroniekzin terugkwam, de vijf vaakste (getoonde regels)
$zinnen = Select-String -Path ($vol + '/hol.log') -Pattern 'HOL kroniek [0-9.]+ \(wacht [0-9.]+\): (.*)$' -ErrorAction SilentlyContinue | ForEach-Object { $_.Matches[0].Groups[1].Value }
$vaak = $zinnen | Group-Object | Where-Object { $_.Count -gt 1 } | Sort-Object Count -Descending | Select-Object -First 5
$samen += "kroniekregels: $(@($zinnen).Count), verschillend $(@($zinnen | Sort-Object -Unique).Count)"
$samen += "vaakst herhaald: $(if ($vaak) { ($vaak | ForEach-Object { "$($_.Count)x '$($_.Name)'" }) -join '; ' } else { 'geen' })"
$samen | Set-Content "$vol/uitslag.txt"
($metingen -join "`n") | Set-Content "$vol/cpu-per-10s.txt"
$samen | ForEach-Object { Write-Output $_ }
if (-not $p.HasExited) { $p.WaitForExit(150000) | Out-Null }
if (-not $p.HasExited) { Stop-Process -Id $p.Id -Force }
Write-Output "===== hol.log (laatste 40 regels)"; Get-Content $log -Tail 40 -ErrorAction SilentlyContinue
if ($env:EXE -eq 'ja') { Copy-Item $glog "$vol/godot.log" -ErrorAction SilentlyContinue }
$fouten = (Get-Content "$vol/uit.txt", "$vol/fout.txt", "$vol/godot.log" -ErrorAction SilentlyContinue | Select-String -CaseSensitive -Pattern "SCRIPT ERROR|Parse Error" | Select-Object -First 20)
Write-Output "===== fouten in de uitvoer: $(@($fouten).Count)"; $fouten | ForEach-Object { Write-Output $_.Line }
$einde = [bool](Select-String -Path $log -Pattern "HOL einde" -Quiet -ErrorAction SilentlyContinue)
Write-Output "===== spel stopte zelf netjes (HOL einde in hol.log): $einde; regels in uit.txt: $(@(Get-Content "$vol/uit.txt" -ErrorAction SilentlyContinue).Count)"
# CPU-grens 1,0: rood alleen bij de .exe (Sanne 17 nov 03:09, poort b); in de editor schommelt de runner te veel, daar een waarschuwing
$cpuRood = ($gem -ge 1.0) -and ($env:EXE -eq 'ja')
if (($gem -ge 1.0) -and ($env:EXE -ne 'ja')) { Write-Output "::warning::CPU gemiddeld $gem boven 1,0 in de editor (alleen waarschuwing; de .exe beslist)" }
# Olaf 26 nov (#194): na ±120 bewaarrondes staan er precies hol.json en hol.vorige.json, geen achtergebleven .tmp
$saves = @(Get-ChildItem (Join-Path $env:APPDATA "Dortus/Het Hol") -Filter "hol*.json*" -Name -ErrorAction SilentlyContinue | Sort-Object)
$saveGoed = (($saves -join ',') -eq 'hol.json,hol.vorige.json')
$saveRegel = "save-bestanden na de duurtest: $($saves -join ', ') (precies hol.json en hol.vorige.json: $saveGoed)"
Write-Output "===== $saveRegel"; Add-Content "$vol/uitslag.txt" $saveRegel
$rood = (-not $einde) -or $dood -or (@($fouten).Count -gt 0) -or $cpuRood -or ($metingen.Count -eq 0) -or (-not $saveGoed)
if ($rood) { Write-Output "== DUURTEST ROOD"; exit 1 } else { Write-Output "== DUURTEST GROEN" }
