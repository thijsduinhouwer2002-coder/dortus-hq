python $GITHUB_WORKSPACE/organisatie/controle/geluid_check.py knee-bend-stem-*.mp4 knee-hook-*.mp4 | tee geluid_check.txt
grep -q "FOUT" geluid_check.txt && ok=1
d=$(ffprobe -v error -show_entries format=duration -of csv=p=0 knee-bend-stem-staand.mp4)
echo "staand $d s" | tee lengtes.txt
python -c "import sys; d=float('$d'); sys.exit(0 if 40.0 <= d <= 60.0 else 1)" || { echo "FOUT: Short is $d s, niet 40 tot 60"; ok=1; }

sha256sum *.mp4 | tee SHA256SUMS.txt
exit $ok
