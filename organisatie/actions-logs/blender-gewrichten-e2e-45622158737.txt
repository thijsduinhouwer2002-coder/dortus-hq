===== e2e (5.3.0) / Direct-zip (gratis met Pro-knoppen op slot, niet voor extensions.blender.org) (laatste 45 van 45 regels) =====
##[group]Run bash e2e/maak_zip.sh gratis
bash e2e/maak_zip.sh gratis
bash e2e/maak_zip.sh direct
python3 -I e2e/toets_direct_zip.py /tmp/click_joints.zip /tmp/click_joints_direct.zip
! unzip -p /tmp/click_joints.zip '*.py' | grep -qiw "pro" || { echo "FOUT: Pro in de extensions-zip"; exit 1; }
/tmp/blender/blender --command extension validate /tmp/click_joints_direct.zip
/tmp/blender/blender -b --factory-startup --python e2e/test_direct.py -- /tmp/click_joints_direct.zip
rm -rf /tmp/cj_direct
/tmp/blender/blender -b --python-expr "import bpy; bpy.context.preferences.view.show_splash=False; bpy.ops.wm.save_userpref(); bpy.ops.wm.save_as_mainfile(filepath='/tmp/start_d.blend')"
mkdir -p "e2e/screenshots/5.3.0"
timeout 240 xvfb-run --auto-servernum --server-args="-screen 0 1920x1080x24" \
  /tmp/blender/blender --gpu-backend opengl /tmp/start_d.blend --python e2e/scherm_direct.py -- "$PWD/e2e/screenshots/5.3.0" /tmp/click_joints_direct.zip 2>&1 | tee /tmp/direct.log || true
grep "\[cj-scherm\] DIRECT" /tmp/direct.log || true
grep -q "DIRECT OK d1-slot" /tmp/direct.log || { echo "FOUT: scherm d1-slot niet goed"; exit 1; }
V=$(grep '^version' click_joints/blender_manifest.toml | cut -d'"' -f2)
mkdir -p "/tmp/direct-zip/5.3.0"
cp /tmp/click_joints_direct.zip "/tmp/direct-zip/5.3.0/click_joints-${V}-direct.zip"
sha256sum /tmp/click_joints_direct.zip | tee "/tmp/direct-zip/5.3.0/click_joints-${V}-direct.sha256"
shell: /usr/bin/bash -e {0}
env:
  PRO_VERSIE: 1.2.9
  LIBGL_ALWAYS_SOFTWARE: 1
  BLENDER_USER_RESOURCES: /tmp/cj_direct
  BLENDER_USER_CONFIG: /tmp/cj_direct/config
  BLENDER_USER_EXTENSIONS: /tmp/cj_direct/extensions
##[endgroup]
